== BibTeX

[source,bibtex]
----
@software{first-post_2026,
  author = {{{AUTHOR_LAST}}, {{AUTHOR_FIRST}}},
  title = {first-post},
  year = {2026},
  url = {https://github.com/metadatastician/first-post},
  license = {MPL-2.0}
}
----
